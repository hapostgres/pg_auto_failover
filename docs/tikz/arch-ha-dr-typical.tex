  %% -----------------------------------------------------------------
  %% "service boundary" box: dashed pastel border, pale tint fill,
  %% bold header label + rule -- compact height, no more than the
  %% content actually needs.
  %% -----------------------------------------------------------------

  \newcommand{\boundarybox}[5]{
    %% #1 x1  #2 y1  #3 x2  #4 y2  #5 border/fill color
    \draw[rounded corners=9pt,draw=#5,very thick,dashed,fill=#5!12]
         (#1,#2) rectangle (#3,#4);
  }

  \newcommand{\boundaryheader}[4]{
    %% #1 x (left)  #2 y (top)  #3 text color  #4 text
    \node[anchor=north west,text=#3,font=\bfseries\Large,inner sep=0pt]
         at (#1,#2) {#4};
  }

  \newcommand{\servicebox}[5]{
    %% #1 x  #2 y  #3 border color  #4 width  #5 text
    \node[rectangle,rounded corners=5pt,draw=#3,thick,fill=white,
          text=stxt,minimum width=#4,minimum height=1.35cm,align=center]
         at (#1,#2) {\normalsize #5};
  }

  %% -- left boundary: High Availability --
  \boundarybox{-12.5}{0.5}{-0.7}{6.7}{haBorder}
  \boundaryheader{-12}{6.3}{haTxt}{High Availability}
  \draw[draw=haBorder,line width=0.6pt] (-12,5.6) -- (-1.2,5.6);

  \servicebox{-6.6}{4.5}{haBorder}{7.4cm}{Patroni}
  \servicebox{-6.6}{1.6}{haBorder}{7.4cm}{repmgr}

  %% -- right boundary: Disaster Recovery + Backups --
  \boundarybox{0.7}{0.5}{12.5}{6.7}{bkBorder}
  \boundaryheader{1.2}{6.3}{bkTxt}{Disaster Recovery + Backups}
  \draw[draw=bkBorder,line width=0.6pt] (1.2,5.6) -- (12,5.6);

  \servicebox{6.6}{4.5}{bkBorder}{7.4cm}{pgBackRest}
  \servicebox{6.6}{1.6}{bkBorder}{7.4cm}{pgBarman}
